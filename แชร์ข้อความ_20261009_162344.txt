local function SmoothStep(a)
a = math.clamp(a, 0, 1)
return a * a * (3 - 2 * a)
end

local function EaseOutCubic(a)
a = math.clamp(a, 0, 1)
return 1 - (1 - a) ^ 3
end

local function WaitFrame()
return RunService.RenderStepped:Wait()
end

local function Animate(duration, callback)
local elapsed = 0
while Running and elapsed < duration do
local dt = WaitFrame()
if not Running then return false end
dt = math.clamp(dt, 0, 1 / 20)
elapsed = math.min(elapsed + dt, duration)
callback(elapsed / duration, elapsed, dt)
end
return Running
end

local function SpeedTrail(startPos, endPos, color)
local dir = endPos - startPos
local length = dir.Magnitude
if length < 0.1 then return end
for i = 1, 6 do
local trail = Instance.new("Part")
trail.Size = Vector3.new(0.15, 0.15, length)
trail.CFrame = CFrame.lookAt(startPos + dir * (i / 7), endPos) * CFrame.new(0, 0, -length / 2)
trail.Anchored = true
trail.CanCollide = false
trail.CanTouch = false
trail.CanQuery = false
trail.Material = Enum.Material.Neon
trail.Color = color or Color3.fromRGB(255, 255, 255)
trail.Transparency = 0.3
trail.Parent = Studio
local tw = TweenService:Create(trail, TweenInfo.new(0.35, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), { Transparency = 1, Size = Vector3.new(0.02, 0.02, length) })
table.insert(Tweens, tw)
tw:Play()
Debris:AddItem(trail, 0.4)
end
end

local function ShakeCamera(intensity, duration)
task.spawn(function()
local start = intensity
local t = 0
while Running and t < duration do
local dt = WaitFrame()
t = t + dt
CameraShakeIntensity = start * (1 - t / duration)
end
CameraShakeIntensity = 0
end)
end

local function RestoreCamera()
if not Camera then return end

Camera.CameraSubject = OriginalCameraSubject
Camera.FieldOfView = OriginalFOV
end

local function Cleanup()
if not Running then return end
Running = false
for _, c in ipairs(Connections) do pcall(function() c:Disconnect() end) end
for _, t in ipairs(Tweens) do pcall(function() t:Cancel() end) end
RestoreCamera()
if Studio.Parent then Studio:Destroy() end
local ui = PlayerGui:FindFirstChild("BaconStudioUI")
if ui then ui:Destroy() end
end

local function ExecuteAnimation()
local baconStart = BaseCFrame * CFrame.new(-7, 3, 0)
local enemyStart = BaseCFrame * CFrame.new(7, 3, 0)

PoseBacon(baconStart, 0, Vector3.one, Vector3.zero, Vector3.zero, Vector3.zero, Vector3.zero, Vector3.zero, Vector3.zero)
PoseEnemy(enemyStart, 0, Vector3.one, Vector3.zero, Vector3.zero, Vector3.zero, Vector3.zero, Vector3.zero, Vector3.zero)

TargetFOV = 65
task.wait(0.8)
if not Running then return end

if not Animate(0.3, function(a)
local s = SmoothStep(a)
PoseBacon(baconStart, -0.5 * s, Vector3.new(1 + 0.14 * s, 1 - 0.24 * s, 1 + 0.1 * s), Vector3.new(14 * s, 0, 0), Vector3.new(-9 * s, 0, 0), Vector3.new(-38 * s, 0, -12 * s), Vector3.new(-38 * s, 0, 12 * s), Vector3.new(38 * s, 0, 0), Vector3.new(38 * s, 0, 0))
PoseEnemy(enemyStart, -0.5 * s, Vector3.new(1 + 0.14 * s, 1 - 0.24 * s, 1 + 0.1 * s), Vector3.new(14 * s, 0, 0), Vector3.new(-9 * s, 0, 0), Vector3.new(-38 * s, 0, -12 * s), Vector3.new(-38 * s, 0, 12 * s), Vector3.new(38 * s, 0, 0), Vector3.new(38 * s, 0, 0))
end) then return end

local g = 55
local jh = 7
local lv = math.sqrt(2 * g * jh)
local fd = 2 * lv / g
local ft = 0
TargetFOV = 58

while Running and ft < fd do
local dt = math.clamp(WaitFrame(), 0, 1 / 30)
ft = math.min(ft + dt, fd)
local y = lv * ft - 0.5 * g * ft * ft
local vy = lv - g * ft
local nv = math.clamp(vy / lv, -1, 1)
local st = math.clamp(math.abs(nv), 0, 1)
local sx = 1 - 0.07 * st
local sy = 1 + 0.14 * st
local sz = 1 - 0.07 * st
local fa = math.clamp(ft / fd, 0, 1)
local flipAngle = math.pi * 2 * fa
local tuck = math.sin(fa * math.pi)
local baconCF = baconStart * CFrame.new(0, y, 0) * CFrame.Angles(flipAngle, 0, 0)
PoseBacon(baconCF, 0, Vector3.new(sx, sy, sz), Vector3.new(24 * tuck, 0, 0), Vector3.new(-9 * tuck, 0, 0), Vector3.new(-115 * tuck - 15, 0, -14), Vector3.new(-115 * tuck - 15, 0, 14), Vector3.new(88 * tuck, 0, 0), Vector3.new(88 * tuck, 0, 0))
end

if not Running then return end

TargetFOV = 70
TriggerShockwave(BaconTorso.Position - Vector3.new(0, 1, 0), 1.2, Color3.fromRGB(0, 200, 255))
TriggerImpact(BaconTorso.Position + Vector3.new(0, -2, 0), Color3.fromRGB(0, 220, 255), 1.3)
ShakeCamera(1.1, 0.45)

if not Animate(0.24, function(a)
local e = 1 - EaseOutCubic(a)
PoseBacon(baconStart, -0.5 * e, Vector3.new(1 + 0.28 * e, 1 - 0.3 * e, 1 + 0.22 * e), Vector3.new(22 * e, 0, 0), Vector3.new(-14 * e, 0, 0), Vector3.new(-65 * e, 0, -10 * e), Vector3.new(-65 * e, 0, 10 * e), Vector3.new(55 * e, 0, 0), Vector3.new(55 * e, 0, 0))
end) then return end

if not Animate(0.4, function(a)
local s = SmoothStep(a)
PoseBacon(baconStart, -0.5 * (1 - s), Vector3.new(1 + 0.28 * (1 - s), 1 - 0.3 * (1 - s), 1 + 0.22 * (1 - s)), Vector3.new(22 * (1 - s), 0, 0), Vector3.new(-14 * (1 - s), 0, 0), Vector3.new(-65 * (1 - s), 0, -10 * (1 - s)), Vector3.new(-65 * (1 - s), 0, 10 * (1 - s)), Vector3.new(55 * (1 - s), 0, 0), Vector3.new(55 * (1 - s), 0, 0))
end) then return end

TargetFOV = 62
task.wait(0.3)
if not Running then return end

if not Animate(0.25, function(a)
local s = SmoothStep(a)
PoseEnemy(enemyStart, -0.4 * s, Vector3.new(1 + 0.15 * s, 1 - 0.25 * s, 1 + 0.12 * s), Vector3.new(24 * s, 0, 0), Vector3.new(-10 * s, 0, 0), Vector3.new(-55 * s, 0, -20 * s), Vector3.new(-55 * s, 0, 20 * s), Vector3.new(50 * s, 0, 0), Vector3.new(50 * s, 0, 0))
end) then return end

task.wait(0.15)
if not Running then return end

local enemyDashFrom = enemyStart
local enemyDashTo = BaseCFrame * CFrame.new(-2.5, 1.2, 0)

SpeedTrail(enemyDashFrom.Position, enemyDashTo.Position, Color3.fromRGB(255, 80, 100))

if not Animate(0.55, function(a)
local s = EaseOutCubic(a)
local rootCF = enemyDashFrom:Lerp(enemyDashTo, s)
PoseEnemy(rootCF, 0, Vector3.new(0.95, 1.06, 0.95), Vector3.new(18, 0, 0), Vector3.new(-8, 0, 0), Vector3.new(-140 + 40 * a, 0, -8), Vector3.new(-140 + 40 * a, 0, 8), Vector3.new(28 - 10 * a, 0, 0), Vector3.new(-15 + 10 * a, 0, 0))
end) then return end

if not Running then return end

TargetFOV = 68
TriggerImpact(EnemyRArm.Position, Color3.fromRGB(255, 90, 120), 1.1)
ShakeCamera(0.7, 0.3)

if not Animate(0.16, function(a)
local s = SmoothStep(a)
PoseEnemy(enemyDashTo, 0, Vector3.one, Vector3.new(24 * s, 0, 0), Vector3.new(-14 * s, 0, 0), Vector3.new(-95 * s, 0, -10 * s), Vector3.new(60 * s, 0, 10 * s), Vector3.new(15 * s, 0, 0), Vector3.new(-15 * s, 0, 0))
end) then return end

if not Animate(0.22, function(a)
local s = SmoothStep(a)
PoseBacon(baconStart, -0.25 * s, Vector3.new(1 + 0.08 * s, 1 - 0.15 * s, 1 + 0.05 * s), Vector3.new(-12 * s, 0, 0), Vector3.new(20 * s, 0, 0), Vector3.new(-20 * s, 0, -30 * s), Vector3.new(65 * s, 0, 30 * s), Vector3.new(10 * s, 0, 0), Vector3.new(-8 * s, 0, 0))
