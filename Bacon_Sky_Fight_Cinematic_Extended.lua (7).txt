local function EaseInOutSine(a)
a = math.clamp(a, 0, 1)
return -(math.cos(math.pi * a) - 1) / 2
end

local function EaseOutQuint(a)
a = math.clamp(a, 0, 1)
return 1 - (1 - a) ^ 5
end

local function WaitFrame()
return RunService.RenderStepped:Wait()
end

local function Animate(duration, callback)
local elapsed = 0
while Running and IsPlaying and elapsed < duration do
local dt = WaitFrame()
if not Running or not IsPlaying then return false end
dt = math.clamp(dt, 0, 1 / 30)
elapsed = math.min(elapsed + dt, duration)
callback(elapsed / duration, elapsed, dt)
end
return Running and IsPlaying
end

local function ShakeCamera(intensity, duration)
task.spawn(function()
local start = intensity
local t = 0
while Running and t < duration do
local dt = WaitFrame()
t = t + dt
CameraShakeIntensity = start * (1 - t / duration) ^ 2
end
CameraShakeIntensity = 0
end)
end

local CamConn = RunService.RenderStepped:Connect(function(dt)
if not Running then return end
local bTorso = BaconTorso.Position
local eTorso = EnemyTorso.Position
local mid = (bTorso + eTorso) * 0.5
local toEnemy = eTorso - bTorso
if toEnemy.Magnitude < 0.1 then toEnemy = Vector3.new(1, 0, 0) end
toEnemy = toEnemy.Unit
local perp = toEnemy:Cross(Vector3.new(0, 1, 0))
if perp.Magnitude < 0.1 then perp = Vector3.new(0, 0, 1) end
perp = perp.Unit
local toPlayer = PlayerOrigin - mid
if toPlayer:Dot(perp) < 0 then perp = -perp end
local fighterGap = (bTorso - eTorso).Magnitude
local orbit = math.sin(os.clock() * 0.62) * 0.32
local cameraSide = (perp * math.cos(orbit) + toEnemy * math.sin(orbit)).Unit
local baseDist = math.clamp(fighterGap * 0.55 + 7.5, 10.5, 17)
local desiredHeight = 3.8 + math.clamp(fighterGap * 0.08, 0, 1.4)
local desiredPos = mid + cameraSide * baseDist + Vector3.new(0, desiredHeight, 0)
local shake = Vector3.new((math.random() - 0.5) * CameraShakeIntensity, (math.random() - 0.5) * CameraShakeIntensity, (math.random() - 0.5) * CameraShakeIntensity)
desiredPos = desiredPos + shake
local sm = 1 - math.exp(-7.5 * math.clamp(dt, 0, 0.1))
camPos = camPos:Lerp(desiredPos, sm)
camFocus = camFocus:Lerp(mid + Vector3.new(0, 2.1, 0), sm)
Camera.CFrame = CFrame.lookAt(camPos, camFocus)
Camera.FieldOfView = Camera.FieldOfView + (TargetFOV - Camera.FieldOfView) * (1 - math.exp(-8 * math.clamp(dt, 0, 0.1)))
end)

Track(CamConn)

local function ExecuteAnimation()
PoseBacon(BaconStart, 0, Vector3.one, Vector3.zero, Vector3.zero, Vector3.zero, Vector3.zero, Vector3.zero, Vector3.zero)
PoseEnemy(EnemyStart, 0, Vector3.one, Vector3.zero, Vector3.zero, Vector3.zero, Vector3.zero, Vector3.zero, Vector3.zero)

TargetFOV = 65
task.wait(0.9)

if not Animate(0.35, function(a)
local s = EaseInOutSine(a)
PoseBacon(BaconStart, -0.5 * s, Vector3.new(1 + 0.15 * s, 1 - 0.25 * s, 1 + 0.12 * s), Vector3.new(15 * s, 0, 0), Vector3.new(-10 * s, 0, 0), Vector3.new(-42 * s, 0, -14 * s), Vector3.new(-42 * s, 0, 14 * s), Vector3.new(42 * s, 0, 0), Vector3.new(42 * s, 0, 0))
PoseEnemy(EnemyStart, -0.5 * s, Vector3.new(1 + 0.15 * s, 1 - 0.25 * s, 1 + 0.12 * s), Vector3.new(15 * s, 0, 0), Vector3.new(-10 * s, 0, 0), Vector3.new(-42 * s, 0, -14 * s), Vector3.new(-42 * s, 0, 14 * s), Vector3.new(42 * s, 0, 0), Vector3.new(42 * s, 0, 0))
end) then return end

task.wait(0.1)
if not Running or not IsPlaying then return end

local g = 55
local jh = 7.5
