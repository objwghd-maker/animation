local function SolveJoints(rootCF, torsoYOffset, squash, torsoRot, headRot, leftArmRot, rightArmRot, leftLegRot, rightLegRot)
    local sx = squash.X
    local sy = squash.Y
    local sz = squash.Z

    Torso.Size = Vector3.new(2 * sx, 2 * sy, sz)
    Torso.CFrame = rootCF * CFrame.new(0, torsoYOffset, 0) * torsoRot

    local torsoCF = Torso.CFrame

    Head.CFrame = torsoCF * CFrame.new(0, sy + 0.5, 0) * headRot
    HairHandle.CFrame = Head.CFrame * CFrame.new(0, 0.45, -0.15)

    LeftArm.CFrame = torsoCF * CFrame.new(-sx - 0.5, 0, 0) * leftArmRot
    RightArm.CFrame = torsoCF * CFrame.new(sx + 0.5, 0, 0) * rightArmRot

    LeftLeg.CFrame = torsoCF * CFrame.new(-0.5 * sx, -sy - 1, 0) * leftLegRot
    RightLeg.CFrame = torsoCF * CFrame.new(0.5 * sx, -sy - 1, 0) * rightLegRot
end

local function Pose(rootCF, y, scale, tr, hr, lar, rar, llr, rlr)
    SolveJoints(
        rootCF,
        y,
        scale,
        CFrame.Angles(math.rad(tr.X), math.rad(tr.Y), math.rad(tr.Z)),
        CFrame.Angles(math.rad(hr.X), math.rad(hr.Y), math.rad(hr.Z)),
        CFrame.Angles(math.rad(lar.X), math.rad(lar.Y), math.rad(lar.Z)),
        CFrame.Angles(math.rad(rar.X), math.rad(rar.Y), math.rad(rar.Z)),
        CFrame.Angles(math.rad(llr.X), math.rad(llr.Y), math.rad(llr.Z)),
        CFrame.Angles(math.rad(rlr.X), math.rad(rlr.Y), math.rad(rlr.Z))
    )
end

local function SmoothStep(a)
    a = math.clamp(a, 0, 1)
    return a * a * (3 - 2 * a)
end

local function EaseOutCubic(a)
    a = math.clamp(a, 0, 1)
    return 1 - (1 - a) ^ 3
end

local function WaitFrame()
    return RunService.RenderStepped:Wait()
end

local function Animate(duration, callback)
    local elapsed = 0

    while Running and elapsed < duration do
        local dt = WaitFrame()
        if not Running then
            return false
        end

        dt = math.clamp(dt, 0, 1 / 20)
        elapsed = math.min(elapsed + dt, duration)

        callback(elapsed / duration, elapsed, dt)
    end

    return Running
end

local function TriggerShockwave(position)
    local ring = Instance.new("Part")
    ring.Name = "LandingShockwave"
    ring.Shape = Enum.PartType.Cylinder
    ring.Size = Vector3.new(0.06, 0.5, 0.5)
    ring.CFrame = CFrame.new(position) * CFrame.Angles(0, 0, math.rad(90))
    ring.Anchored = true
    ring.CanCollide = false
    ring.CanTouch = false
    ring.CanQuery = false
    ring.Material = Enum.Material.Neon
    ring.Color = Color3.fromRGB(0, 200, 255)
    ring.Transparency = 0.1
    ring.Parent = Studio

    local tween = TweenService:Create(
        ring,
        TweenInfo.new(0.5, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
        {
            Size = Vector3.new(0.06, 12, 12),
            Transparency = 1
        }
    )

    table.insert(Tweens, tween)
    tween:Play()
    Debris:AddItem(ring, 0.6)
end

local function RestoreCamera()
    if not Camera then
        return
    end

    Camera.CameraType = OriginalCameraType
    Camera.CameraSubject = OriginalCameraSubject
    Camera.FieldOfView = OriginalFOV
end

local function Cleanup()
    if not Running then
        return
    end

    Running = false

    for _, connection in ipairs(Connections) do
        pcall(function()
            connection:Disconnect()
        end)
    end

    for _, tween in ipairs(Tweens) do
        pcall(function()
            tween:Cancel()
        end)
    end

    RestoreCamera()

    if Studio.Parent then
        Studio:Destroy()
    end

    local ui = PlayerGui:FindFirstChild("BaconStudioUI")
    if ui then
        ui:Destroy()
    end
end

local cameraPosition = BaseCFrame.Position + Vector3.new(0, 4.5, -9)
local cameraFocus = BaseCFrame.Position + Vector3.new(0, 3, 0)

local CameraConnection = RunService.RenderStepped:Connect(function(dt)
    if not Running or not Root.Parent then
        return
    end

    local targetFocus = Torso.Position + Vector3.new(0, 0.4, 0)
    local shake = Vector3.new(
        (math.random() - 0.5) * CameraShakeIntensity,
        (math.random() - 0.5) * CameraShakeIntensity,
        0
    )

    local desiredPosition = targetFocus + Vector3.new(0, 1.2, -8.5) + shake
    local smoothing = 1 - math.exp(-5 * math.clamp(dt, 0, 0.1))

    cameraPosition = cameraPosition:Lerp(desiredPosition, smoothing)
    cameraFocus = cameraFocus:Lerp(targetFocus, smoothing)

    Camera.CFrame = CFrame.lookAt(cameraPosition, cameraFocus)
    Camera.FieldOfView = Camera.FieldOfView
        + (TargetFOV - Camera.FieldOfView)
        * (1 - math.exp(-4 * math.clamp(dt, 0, 0.1)))
end)

TrackConnection(CameraConnection)

local function ExecuteAnimation()
    local startCF = BaseCFrame * CFrame.new(0, 3, 0)

    Pose(
        startCF, 0, Vector3.one,
        Vector3.zero, Vector3.zero,
        Vector3.zero, Vector3.zero,
        Vector3.zero, Vector3.zero
    )

    task.wait(0.5)
    if not Running then return end

    if not Animate(0.32, function(alpha)
        local a = SmoothStep(alpha)
        local y = -0.45 * a
        local scale = Vector3.new(
            1 + 0.12 * a,
            1 - 0.22 * a,
            1 + 0.10 * a
        )

        Pose(
            startCF, y, scale,
            Vector3.new(12 * a, 0, 0),
            Vector3.new(-8 * a, 0, 0),
            Vector3.new(-35 * a, 0, -12 * a),
            Vector3.new(-35 * a, 0, 12 * a),
            Vector3.new(35 * a, 0, 0),
            Vector3.new(35 * a, 0, 0)
        )
    end) then return end

    local gravity = 65
    local jumpHeight = 5.5
    local launchVelocity = math.sqrt(2 * gravity * jumpHeight)
    local flightDuration = 2 * launchVelocity / gravity
    local flightTime = 0

    TargetFOV = 59

    while Running and flightTime < flightDuration do
        local dt = math.clamp(WaitFrame(), 0, 1 / 30)
        flightTime = math.min(flightTime + dt, flightDuration)

        local t = flightTime
        local verticalDisplacement = launchVelocity * t - 0.5 * gravity * t * t
        local verticalVelocity = launchVelocity - gravity * t

        local normalizedVelocity = math.clamp(verticalVelocity / launchVelocity, -1, 1)
        local stretch = math.clamp(math.abs(normalizedVelocity), 0, 1)

        local sx = 1 - 0.06 * stretch
        local sy = 1 + 0.12 * stretch
        local sz = 1 - 0.06 * stretch

        local rootCF = startCF * CFrame.new(0, verticalDisplacement, 0)
        local airborneAlpha = math.clamp(t / flightDuration, 0, 1)

        local torsoPitch = -8 + 16 * airborneAlpha
        local armAngle = -65 + 15 * airborneAlpha
        local legAngle = 18 + 12 * airborneAlpha

        if verticalVelocity < 0 then
            torsoPitch = torsoPitch - 6 * stretch
        end

        Pose(
            rootCF, 0, Vector3.new(sx, sy, sz),
            Vector3.new(torsoPitch, 0, 0),
            Vector3.new(-torsoPitch * 0.35, 0, 0),
            Vector3.new(armAngle, 0, -12),
            Vector3.new(armAngle, 0, 12),
            Vector3.new(legAngle, 0, 0),
            Vector3.new(legAngle, 0, 0)
        )
    end

    if not Running then return end

    TargetFOV = 65
    CameraShakeIntensity = 0.65

    TriggerShockwave(Torso.Position - Vector3.new(0, 1, 0))

    task.spawn(function()
        for i = 1, 10 do
            if not Running then return end
            CameraShakeIntensity = 0.65 * (1 - i / 10)
            task.wait(0.025)
        end
        CameraShakeIntensity = 0
    end)

    if not Animate(0.22, function(alpha)
        local a = 1 - EaseOutCubic(alpha)
        local y = -0.35 * a
        local scale = Vector3.new(
            1 + 0.22 * a,
            1 - 0.25 * a,
            1 + 0.18 * a
        )

        Pose(
            startCF, y, scale,
            Vector3.new(18 * a, 0, 0),
            Vector3.new(-12 * a, 0, 0),
            Vector3.new(-55 * a, 0, -8 * a),
            Vector3.new(-55 * a, 0, 8 * a),
            Vector3.new(45 * a, 0, 0),
            Vector3.new(45 * a, 0, 0)
        )
    end) then return end

    Animate(0.28, function(alpha)
        local a = SmoothStep(alpha)

        local y = -0.35 * (1 - a)
        local scale = Vector3.new(
            1 + 0.22 * (1 - a),
            1 - 0.25 * (1 - a),
            1 + 0.18 * (1 - a)
        )

        Pose(
            startCF, y, scale,
            Vector3.new(18 * (1 - a), 0, 0),
            Vector3.new(-12 * (1 - a), 0, 0),
            Vector3.new(-55 * (1 - a), 0, -8 * (1 - a)),
            Vector3.new(-55 * (1 - a), 0, 8 * (1 - a)),
            Vector3.new(45 * (1 - a), 0, 0),
            Vector3.new(45 * (1 - a), 0, 0)
        )
    end)

    if not Running then return end

    Pose(
        startCF, 0, Vector3.one,
        Vector3.zero, Vector3.zero,
        Vector3.zero, Vector3.zero,
        Vector3.zero, Vector3.zero
    )
end

task.spawn(function()
    local success, err = pcall(ExecuteAnimation)

    if not success then
        warn("Bacon animation error:", err)
    end
end)
