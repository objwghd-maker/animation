Camera.CameraType = Enum.CameraType.Scriptable
Camera.FieldOfView = 65

local CamDistance = 9.5
local CamHeight = 2.4
local ShakeAmount = 0

local cameraConn
cameraConn = RunService.RenderStepped:Connect(function()
    if not Root or not Root.Parent then
        if cameraConn then cameraConn:Disconnect() end
        Camera.CameraType = Enum.CameraType.Custom
        return
    end
    
    local shakeX = (math.random() - 0.5) * ShakeAmount
    local shakeY = (math.random() - 0.5) * ShakeAmount
    
    local focusPos = Root.Position + Vector3.new(0, 1.2, 0)
    local camPos = focusPos + Vector3.new(shakeX, CamHeight + shakeY, CamDistance)
    
    Camera.CFrame = CFrame.lookAt(camPos, focusPos)
end)

local function applyPose(pose, duration, easingStyle, easingDirection)
    local tweenInfo = TweenInfo.new(duration, easingStyle or Enum.EasingStyle.Quad, easingDirection or Enum.EasingDirection.Out)
    
    if pose.Root then TweenService:Create(RootJoint, tweenInfo, {C0 = pose.Root}):Play() end
    if pose.Neck then TweenService:Create(Neck, tweenInfo, {C0 = pose.Neck}):Play() end
    if pose.LArm then TweenService:Create(LeftShoulder, tweenInfo, {C0 = pose.LArm}):Play() end
    if pose.RArm then TweenService:Create(RightShoulder, tweenInfo, {C0 = pose.RArm}):Play() end
    if pose.LLeg then TweenService:Create(LeftHip, tweenInfo, {C0 = pose.LLeg}):Play() end
    if pose.RLeg then TweenService:Create(RightHip, tweenInfo, {C0 = pose.RLeg}):Play() end
    
    task.wait(duration)
end

    local Wave = Instance.new("Part")

    Wave.Transparency = 0.2
    Wave.Parent = Studio
    
    local Mesh = Instance.new("SpecialMesh")
    Mesh.MeshType = Enum.MeshType.Cylinder
    Mesh.Parent = Wave
    
    TweenService:Create(Wave, TweenInfo.new(0.4, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
        Size = Vector3.new(16, 0.05, 16),
        Transparency = 1
    }):Play()
    
    Debris:AddItem(Wave, 0.45)
end

local function playMasterAnimationSequence()

        LArm = CFrame.new(-1.5, 0.5, 0),

        RLeg = CFrame.new(0.5, -1, 0)
